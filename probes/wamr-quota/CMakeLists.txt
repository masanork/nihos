cmake_minimum_required(VERSION 3.14)
project(nihos_wamr_quota C ASM)
if(NOT DEFINED WAMR_ROOT_DIR)
  message(FATAL_ERROR "Set WAMR_ROOT_DIR")
endif()
string(TOLOWER "${CMAKE_HOST_SYSTEM_NAME}" WAMR_BUILD_PLATFORM)
if(APPLE)
  add_definitions(-DBH_PLATFORM_DARWIN)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)")
  set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)")
  set(WAMR_BUILD_TARGET X86_64)
else()
  message(FATAL_ERROR "Unsupported host")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_BUILD_ALLOC_WITH_USAGE 1)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
add_executable(wamr-quota main.c)
set_target_properties(wamr-quota PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_options(wamr-quota PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_include_directories(wamr-quota PRIVATE ${WAMR_ROOT_DIR}/core/iwasm/include)
target_link_libraries(wamr-quota PRIVATE wamr m pthread)

if(NIHOS_WAMR_BUDDY)
  if(NOT EXISTS "${NIHOS_ARENA_OBJ}")
    message(FATAL_ERROR "Set NIHOS_ARENA_OBJ to the compiled Zig arena object")
  endif()
  target_compile_definitions(wamr-quota PRIVATE NIHOS_WAMR_BUDDY=1)
  set_source_files_properties("${NIHOS_ARENA_OBJ}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  target_sources(wamr-quota PRIVATE "${NIHOS_ARENA_OBJ}")
endif()

idf_component_register(SRCS "main.c" REQUIRES esp_driver_sdmmc sdmmc esp_driver_gpio esp_hw_support)
set(NIHOS_ORACLE_WIDTH 1 CACHE STRING "SDMMC reference bus width")
set(NIHOS_ORACLE_KHZ 400 CACHE STRING "SDMMC reference requested clock")
set(NIHOS_ORACLE_SAMPLES 0 CACHE STRING "Distributed read-only sample count")
if(NOT "${NIHOS_ORACLE_WIDTH}:${NIHOS_ORACLE_KHZ}" MATCHES "^(1:400|4:400|4:20000)$")
    message(FATAL_ERROR "Unsupported read-only oracle transport")
endif()
if(NOT NIHOS_ORACLE_SAMPLES MATCHES "^(0|1024)$")
    message(FATAL_ERROR "Unsupported reference sample count")
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE NIHOS_ORACLE_WIDTH=${NIHOS_ORACLE_WIDTH} NIHOS_ORACLE_KHZ=${NIHOS_ORACLE_KHZ} NIHOS_ORACLE_SAMPLES=${NIHOS_ORACLE_SAMPLES})

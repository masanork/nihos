cmake_minimum_required(VERSION 3.16)
project(nihos_graphics_memory_probe C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PROBE_SANITIZERS "Instrument host libraries and probes" ON)
option(PROBE_LINK_ORACLE "Build memory-stream host/RV32 link oracle" OFF)
option(PROBE_TASK_ORACLE "Use native trap assembly and common task runtime in ISA fixture" OFF)
if(PROBE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Os -ffunction-sections -fdata-sections)
if(PROBE_LINK_ORACLE AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
  add_compile_options(-fstack-usage -fcallgraph-info=su)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(dep HARFBUZZ ZLIB PNG BZIP2 BROTLI)
  set(FT_DISABLE_${dep} ON CACHE BOOL "" FORCE)
endforeach()
foreach(dep CORETEXT GLIB GOBJECT ICU CAIRO GRAPHITE2)
  set(HB_HAVE_${dep} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(part UTILS SUBSET VECTOR GPU RASTER)
  set(HB_BUILD_${part} OFF CACHE BOOL "" FORCE)
endforeach()
set(HB_BUILD_GPU_DEMO OFF CACHE STRING "" FORCE)
set(HB_HAVE_FREETYPE ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_PNG TRUE)
add_subdirectory("${PROBE_SOURCES}/freetype" ft)
add_subdirectory("${PROBE_SOURCES}/harfbuzz" hb)
target_compile_definitions(harfbuzz PRIVATE
  HB_CONFIG_OVERRIDE_H="${CMAKE_CURRENT_SOURCE_DIR}/hb_config.h"
  hb_malloc_impl=study_malloc hb_calloc_impl=study_calloc
  hb_realloc_impl=study_realloc hb_free_impl=study_free)
if(PROBE_LINK_ORACLE)
  if(NOT PROBE_ALLOCATION_RECORDS)
    set(PROBE_ALLOCATION_RECORDS 256)
  endif()
  if(NOT PROBE_ALLOCATION_RECORDS MATCHES "^(256|512|1024)$")
    message(FATAL_ERROR "link oracle allocation records must be 256, 512 or 1024")
  endif()
  get_target_property(ft_sources freetype SOURCES)
  list(FILTER ft_sources EXCLUDE REGEX "(^|/)ftsystem\\.c$")
  set_property(TARGET freetype PROPERTY SOURCES "${ft_sources}")
  target_sources(freetype PRIVATE ft_noio.c)
  target_compile_definitions(freetype PRIVATE FT_CONFIG_OPTIONS_H="${CMAKE_CURRENT_SOURCE_DIR}/ft_noenv.h")
  if(PROBE_FT_RENDER_POOL_BYTES)
    if(NOT PROBE_FT_RENDER_POOL_BYTES MATCHES "^(2048|16384)$")
      message(FATAL_ERROR "link oracle render pool must be 2048 or 16384")
    endif()
    target_compile_definitions(freetype PRIVATE PROBE_FT_RENDER_POOL_BYTES=${PROBE_FT_RENDER_POOL_BYTES})
  endif()
  add_executable(font-link-oracle "${PROBE_GENERATED}/link_oracle.cpp" lazy_pool.cpp)
  target_include_directories(font-link-oracle PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(font-link-oracle PRIVATE NIHOS_GRAPHICS_BUDDY_BACKING=1)
  target_compile_definitions(font-link-oracle PRIVATE NIHOS_GRAPHICS_ALLOCATION_RECORDS=${PROBE_ALLOCATION_RECORDS})
  target_compile_options(font-link-oracle PRIVATE -fno-exceptions -fno-rtti)
  target_link_libraries(font-link-oracle PRIVATE harfbuzz freetype "${PROBE_BUDDY_OBJECT}" m)
  if(CMAKE_SYSTEM_NAME STREQUAL "Generic")
    target_compile_definitions(font-link-oracle PRIVATE PROBE_RV32=1)
    set(oracle_layout "${CMAKE_CURRENT_SOURCE_DIR}/rv32_link.ld")
    if(PROBE_TASK_ORACLE)
      enable_language(ASM)
      set(oracle_layout "${CMAKE_CURRENT_SOURCE_DIR}/worker/link.ld")
      target_compile_definitions(font-link-oracle PRIVATE PROBE_TASK_ORACLE=1 NIH_NATIVE_FLOATING=0 NIH_NATIVE_DISPLAY_SCANOUT=0 NIH_NATIVE_REV3=1)
      target_sources(font-link-oracle PRIVATE ../../src/platform/native/start.S)
      target_compile_options(font-link-oracle PRIVATE
        $<$<COMPILE_LANGUAGE:ASM>:-march=rv32imac_zicsr_zifencei>
        $<$<COMPILE_LANGUAGE:ASM>:-mabi=ilp32>)
      target_include_directories(font-link-oracle PRIVATE ../../src/arch/riscv32 ../../src/kernel)
      target_link_libraries(font-link-oracle PRIVATE "${PROBE_WORKER_OBJECT}")
    endif()
    target_link_options(font-link-oracle PRIVATE -nostartfiles --specs=nosys.specs
      "-Wl,-T,${oracle_layout}" -Wl,--gc-sections
      "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/oracle.map")
  else()
    target_sources(font-link-oracle PRIVATE link_host_main.cpp)
  endif()
  return()
endif()
add_library(lazy_pool STATIC lazy_pool.cpp)
target_compile_options(lazy_pool PRIVATE -Wall -Wextra -Wconversion -Werror)
add_executable(pool-test pool_test.cpp)
target_link_libraries(pool-test PRIVATE lazy_pool)
add_executable(font-probe font_probe.cpp)
target_compile_options(font-probe PRIVATE -Wall -Wextra -Wconversion -Werror -fno-exceptions -fno-rtti)
target_link_libraries(font-probe PRIVATE lazy_pool harfbuzz freetype)
if(PROBE_BUDDY_OBJECT)
  add_library(lazy_pool_buddy STATIC lazy_pool.cpp buddy_driver.cpp)
  target_compile_definitions(lazy_pool_buddy PRIVATE NIHOS_GRAPHICS_BUDDY_BACKING=1)
  target_compile_options(lazy_pool_buddy PRIVATE -Wall -Wextra -Wconversion -Werror)
  add_executable(backing-test backing_test.cpp)
  target_link_libraries(backing-test PRIVATE lazy_pool_buddy "${PROBE_BUDDY_OBJECT}")
  add_executable(pool-test-buddy pool_test.cpp)
  target_compile_definitions(pool-test-buddy PRIVATE NIHOS_GRAPHICS_BUDDY_BACKING=1)
  target_link_libraries(pool-test-buddy PRIVATE lazy_pool_buddy "${PROBE_BUDDY_OBJECT}")
  add_executable(font-probe-buddy font_probe.cpp)
  target_compile_definitions(font-probe-buddy PRIVATE NIHOS_GRAPHICS_BUDDY_BACKING=1)
  target_compile_options(font-probe-buddy PRIVATE -Wall -Wextra -Wconversion -Werror -fno-exceptions -fno-rtti)
  target_link_libraries(font-probe-buddy PRIVATE lazy_pool_buddy harfbuzz freetype "${PROBE_BUDDY_OBJECT}")
endif()

cmake_minimum_required(VERSION 3.14)
project(nihos_wamr_probe C ASM)
if(NOT DEFINED WAMR_ROOT_DIR)
  message(FATAL_ERROR "Set WAMR_ROOT_DIR")
endif()
string(TOLOWER "${CMAKE_HOST_SYSTEM_NAME}" WAMR_BUILD_PLATFORM)
if(APPLE)
  add_definitions(-DBH_PLATFORM_DARWIN)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)")
  set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)")
  set(WAMR_BUILD_TARGET X86_64)
else()
  message(FATAL_ERROR "Unsupported host CPU")
endif()
set(WAMR_BUILD_INTERP 1)
if(NOT DEFINED NIHOS_WAMR_FAST)
  set(NIHOS_WAMR_FAST 0)
endif()
set(WAMR_BUILD_FAST_INTERP ${NIHOS_WAMR_FAST})
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
add_executable(nihos-wamr-host main.c)
target_compile_options(nihos-wamr-host PRIVATE -UNDEBUG)
target_include_directories(nihos-wamr-host PRIVATE ${WAMR_ROOT_DIR}/core/iwasm/include)
target_link_libraries(nihos-wamr-host PRIVATE wamr m pthread)
if(DEFINED NIHOS_BOOT_LIBRARY)
  add_executable(nihos-wamr-journal main.c)
  target_compile_definitions(nihos-wamr-journal PRIVATE NIHOS_JOURNAL_ADMISSION=1)
  target_compile_options(nihos-wamr-journal PRIVATE -UNDEBUG)
  target_include_directories(nihos-wamr-journal PRIVATE ${WAMR_ROOT_DIR}/core/iwasm/include)
  target_link_libraries(nihos-wamr-journal PRIVATE wamr ${NIHOS_BOOT_LIBRARY} m pthread)
endif()
